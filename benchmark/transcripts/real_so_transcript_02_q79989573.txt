 manual-sign.sh's contents: 

```
#!/bin/bash
set -eo pipefail

APP_PATH="${1:-./out/myApp-darwin-arm64/myApp.app}"
IDENTITY="${APPLE_CODESIGN_IDENTITY:-}"
PLIST="./build/entitlements.mas.plist"
INHERIT="./build/entitlements.mas.inherit.plist"

if [[ -z "$IDENTITY" || "$IDENTITY" == "-" ]]; then
 echo "ERROR: ad-hoc signing is disabled for sandbox testing."
 echo "Set APPLE_CODESIGN_IDENTITY to a real Apple Development identity."
 echo "Example:"
 echo " APPLE_CODESIGN_IDENTITY='Apple Development: Your Name (TEAMID)' npm run package:adhoc"
 exit 2
fi

if ! security find-identity -v -p codesigning | grep -Fq "$IDENTITY"; then
 echo "ERROR: signing identity not found in keychain search list:"
 echo " $IDENTITY"
 echo "Run: security find-identity -v -p codesigning"
 exit 3
fi

if [[ ! -d "$APP_PATH" ]]; then
 echo "App bundle not found: $APP_PATH"
 echo "Usage: APPLE_CODESIGN_IDENTITY='Apple Development: Name (TEAMID)' bash ./manual-sign.sh [path/to/App.app]"
 exit 1
fi

echo "=== Phase 1: Signing Vendor Frameworks ==="
 codesign --force --verbose --timestamp=none --options runtime --sign "$IDENTITY" "$APP_PATH/Contents/Frameworks/Mantle.framework"
 codesign --force --verbose --timestamp=none --options runtime --sign "$IDENTITY" "$APP_PATH/Contents/Frameworks/ReactiveObjC.framework"
 codesign --force --verbose --timestamp=none --options runtime --sign "$IDENTITY" "$APP_PATH/Contents/Frameworks/Squirrel.framework"

echo "=== Phase 2: Signing Electron Core Framework ==="
# Core framework needs to be signed at its specific binary version path
codesign --force --verbose --timestamp=none --options runtime --sign "$IDENTITY" "$APP_PATH/Contents/Frameworks/Electron Framework.framework/Versions/A/Electron Framework"
codesign --force --verbose --timestamp=none --options runtime --sign "$IDENTITY" "$APP_PATH/Contents/Frameworks/Electron Framework.framework"

echo "=== Phase 3: Signing App Helper Sub-processes ==="
# Helpers handle sandbox rendering and must strictly receive the inherit plist
codesign --force --verbose --timestamp=none --options runtime --entitlements "$INHERIT" --sign "$IDENTITY" "$APP_PATH/Contents/Frameworks/myApp Helper (GPU).app"
codesign --force --verbose --timestamp=none --options runtime --entitlements "$INHERIT" --sign "$IDENTITY" "$APP_PATH/Contents/Frameworks/myApp Helper (Plugin).app"
codesign --force --verbose --timestamp=none --options runtime --entitlements "$INHERIT" --sign "$IDENTITY" "$APP_PATH/Contents/Frameworks/myApp Helper (Renderer).app"
codesign --force --verbose --timestamp=none --options runtime --entitlements "$INHERIT" --sign "$IDENTITY" "$APP_PATH/Contents/Frameworks/myApp Helper.app"

echo "=== Phase 4: Signing Main Binary and App Bundle ==="
# The main app bundle binds the configuration together and receives the master plist
codesign --force --verbose --timestamp=none --options runtime --entitlements "$PLIST" --sign "$IDENTITY" "$APP_PATH/Contents/MacOS/myApp"
codesign --force --verbose --timestamp=none --options runtime --entitlements "$PLIST" --sign "$IDENTITY" "$APP_PATH"

echo "=== Phase 5: Verifying Structural Integrity ==="
codesign --verify --deep --strict --verbose=2 "$APP_PATH"
echo "=== App Entitlements ==="
codesign -d --entitlements :- "$APP_PATH" 2>&1
echo "=== Renderer Helper Entitlements ==="
codesign -d --entitlements :- "$APP_PATH/Contents/Frameworks/myApp Helper (Renderer).app" 2>&1

```
